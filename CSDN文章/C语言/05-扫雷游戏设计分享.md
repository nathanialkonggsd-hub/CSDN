# 前言
本文衔接上文函数与数组内容，以制作扫雷小游戏的形式对知识进行具象化的体现与应用。对于初识函数与数组的读者们，可以先阅读<扫雷游戏设计思路>与<主函数--显示游戏界面>，再按顺序阅读以更好地理解思路。
### 扫雷游戏设计思路

扫雷游戏的核心在于玩家交互、地图生成、地雷布置、游戏逻辑判断。设计时需要将整个游戏分解为多个模块，每个模块通过函数实现。

地图采用二维数组表示，一个数组用于存储地雷位置，另一个数组用于显示玩家看到的界面。地雷布置使用随机数生成，确保每次游戏地雷位置不同。

### 初始化游戏地图

定义两个二维数组，一个用于存储地雷分布（`mineMap`），另一个用于显示玩家操作后的界面（`showMap`）。初始化时将所有位置设为未探索状态。

```c
#define ROW 9
#define COL 9
#define MINES 10

char mine[ROW][COL];
char show[ROW][COL];

void Initboard() {
    for (int i = 0; i < ROW; i++) {
        for (int j = 0; j < COL; j++) {
            mine[i][j] = '0';
            show[i][j] = '*';
        }
    }
}
```
但在排雷的过程中，我们需要知道周围一圈雷的个数，但当排查位置为边界时，上述代码会出现越界问题。
优化如下：

```c
#define ROW 9
#define COL 9
#define ROWS ROW+2
#define COLS COL+2
#define COUNT 10

char mine[ROWS][COLS];
char show[ROWS][COLS];

void initMap() {
    for (int i = 0; i < ROWS; i++) {
        for (int j = 0; j < COLS; j++) {
            mine[i][j] = '0';
            show[i][j] = '*';
        }
    }
}
```

### 随机布置地雷

使用随机数生成地雷位置，确保生成的位置不重复且数量符合设定。地雷在`mineMap`中用`'1'`表示。

```c
#include <stdlib.h>
#include <time.h>

void SetMine() {
    srand((unsigned int)time(NULL));
    int count = 0;
    while (count < MINE) {
        int x = rand() % ROW+1;
        int y = rand() % COL+1;
        if (mine[x][y] == '0') {
            mine[x][y] = '1';
            count++;
        }
    }
}
```

### 计算周围地雷数量

遍历每个非地雷格子，统计周围8个格子中的地雷数量，并将结果存入`mineMap`中。使用递归实现空白区域的自动展开。

```c
int Getminecount(int x, int y) {
    int count = 0;
    for (int i = -1; i <= 1; i++) {
        for (int j = -1; j <= 1; j++) {
            int nx = x + i;
            int ny = y + j;
            if (nx >= 0 && nx < ROWS && ny >= 0 && ny < COLS && mine[nx][ny] == '1') {
                count++;
            }
        }
    }
    return count;
}

void Expand(char mine[][100], char show[][100], int x, int y, int r, int c)
{
    int i, j;
    if (x < 1 || x > r || y < 1 || y > c)
        return;
    if (show[x][y] != '*')
        return;
    if (mine[x][y] == '1')
        return;
    
    int count = Getminecount(mine, x, y);
    show[x][y] = count + '0';
    
    if (count == 0) {
        for (i = x - 1; i <= x + 1; i++) {
            for (j = y - 1; j <= y + 1; j++) {
                if (i >= 1 && i <= r && j >= 1 && j <= c && show[i][j] == '*')
                    Expand(mine, show, i, j, r, c);
            }
        }
    }
}
```

### 玩家输入与游戏逻辑

玩家通过输入坐标来探索格子，如果踩到地雷游戏结束，否则显示周围地雷数量。排完则游戏结束。

```c
void Findmine(char mine[][100], char show[][100], int r, int c, int count)
{
    int x, y;
    int op;
    int win = 0;
    int ROWS = r + 2;
    int COLS = c + 2;
    
    while (win < r * c - count) {
        printf("请选择操作: 1.排查 2.标记/取消标记\n");
        if (scanf_s("%d", &op) != 1 || (op != 1 && op != 2)) {
            printf("输入错误，请输入1或2！\n");
            while (getchar() != '\n');
            continue;
        }
        
        if (op == 2) {
            printf("请输入要标记的坐标(1-%d, 1-%d):", r, c);
            if (scanf_s("%d%d", &x, &y) != 2 || x < 1 || x > r || y < 1 || y > c) {
                printf("坐标非法，请输入1到%d之间的数字！\n", r);
                while (getchar() != '\n');
                continue;
            }
            Flagmine(show, x, y);
        } else {
            printf("请输入排查坐标(1-%d, 1-%d):", r, c);
            if (scanf_s("%d%d", &x, &y) != 2 || x < 1 || x > r || y < 1 || y > c) {
                printf("坐标非法，请输入1到%d之间的数字！\n", r);
                while (getchar() != '\n');
                continue;
            }
            
            if (show[x][y] == '!' || show[x][y] == '*') {
                if (mine[x][y] == '1') {
                    printf("很遗憾，你被炸死了！\n");
                    int i, j;
                    for (i = 1; i <= r; i++) {
                        for (j = 1; j <= c; j++) {
                            if (mine[i][j] == '1')
                                show[i][j] = '@';
                        }
                    }
                    Displayboard(show, ROWS, COLS, 1, 1);
                    return;
                } else {
                    Expand(mine, show, x, y, r, c);
                    win = 0;
                    int i, j;
                    for (i = 1; i <= r; i++) {
                        for (j = 1; j <= c; j++) {
                            if (show[i][j] >= '0' && show[i][j] <= '8')
                                win++;
                        }
                    }
                    Displayboard(show, ROWS, COLS, 1, 1);
                }
            } else {
                printf("该位置已被排查，请重新输入坐标！\n");
            }
        }
    }
    
    if (win >= r * c - count) {
        printf("恭喜你！你不怕危险，敢于排除万难取得胜利！\n");
        int i, j;
        for (i = 1; i <= r; i++) {
            for (j = 1; j <= c; j++) {
                if (mine[i][j] == '1')
                    show[i][j] = '!';
            }
        }
        Displayboard(show, ROWS, COLS, 1, 1);
    }
}
```

```c

/* 
 * 函数名: getDiffChoice
 * 功能: 获取用户选择的游戏难度
 * 返回值: 用户选择的难度编号 (1-简单, 2-中等, 3-困难)

int getDiffChoice() {
    int choice;

            while (getchar() != '\n');
        } else {
            return choice;

/* 
 * 函数名: game
 * 功能: 游戏主函数，实现扫雷游戏的核心逻辑
 *       1. 根据选择的难度初始化游戏参数
 *       2. 分配内存创建雷阵和显示阵
 *       3. 布置地雷并开始游戏
 *       4. 计时并统计游戏用时
 */
void game()
{
    Difficulty diff;
    int choice = getDiffChoice();
    
    switch (choice) {
    case 1:
        diff.row = 9;
        diff.col = 9;
        diff.count = 10;
        break;
    case 2:
        diff.row = 16;
        diff.col = 16;
        diff.count = 40;
        break;
    case 3:
        diff.row = 30;
        diff.col = 16;
        diff.count = 99;
        break;
    }

    int ROW = diff.row;
    int COL = diff.col;
    int ROWS = ROW + 2;
    int COLS = COL + 2;
    int COUNT = diff.count;

    char (*mine)[100] = (char(*)[100])malloc(sizeof(char) * 100 * ROWS);
    char (*show)[100] = (char(*)[100])malloc(sizeof(char) * 100 * ROWS);
    
    srand((unsigned int)time(NULL));
    
    Initboard(mine, ROWS, COLS, '0');
    Initboard(show, ROWS, COLS, '*');
    Setmine(mine, ROWS, COLS, COUNT);
    
    printf("开始游戏吧！\n");
    Displayboard(show, ROWS, COLS, 1, 1);
    
    clock_t start = clock();
    Findmine(mine, show, ROW, COL, COUNT);
    clock_t end = clock();
    double time_used = ((double)(end - start)) / CLOCKS_PER_SEC;
    
    printf("用时: %.2f 秒\n", time_used);
    
    free(mine);
    free(show);
}

/* 
 * 函数名: Initboard
 * 功能: 初始化二维数组，将所有元素设置为指定值
 * 参数: 
 *   board - 要初始化的二维数组
 *   r     - 数组行数
 *   c     - 数组列数
 *   set   - 要设置的值('0'初始化雷阵, '*'初始化显示阵)
 */
void Initboard(char board[][100], int r, int c, char set)
{
    int i, j;
    for (i = 0; i < r; i++)
        for (j = 0; j < c; j++)
            board[i][j] = set;

/*

/* 
 * 函数名: Setmine
 * 功能: 随机布置地雷
 * 参数:
 *   board - 雷阵数组
 *   r     - 数组行数
 *   c     - 数组列数
 *   count - 要布置的雷数
 */
void Setmine(char board[][100], int r, int c, int count)
{
    int x, y;
    int ROW = r - 2;
    int COL = c - 2;
    while (count) {
        x = rand() % ROW + 1;
        y = rand() % COL + 1;
        if (board[x][y] == '0') {
            board[x][y] = '1';
            count--;
        }
    }
}

/* 
 * 函数名: Getminecount
 * 功能: 计算指定坐标周围8个格子中的雷数
 * 参数:
 *   mine - 雷阵数组
 *   x    - 要查询的行坐标
 *   y    - 要查询的列坐标
 * 返回值: 周围8个格子中雷的数量(0-8)
 */
int Getminecount(char mine[][100], int x, int y)
{
    int count = 0;
    int i, j;
    for (i = x - 1; i <= x + 1; i++) {
        for (j = y - 1; j <= y + 1; j++) {
            if (mine[i][j] == '1')
                count++;
        }
    }
    return count;
}

/* 
 * 函数名: Expand (新增功能1: 展开周围一片)

 *   y    - 当前排查的列坐标

void Expand(char mine[][100], char show[][100], int x, int y, int r, int c)

                if (i >= 1 && i <= r && j >= 1 && j <= c && show[i][j] == '*')

            }
        }
    }
}

                    }
                    Displayboard(show, ROWS, COLS, 1, 1);
                    return;
                } else {
                    Expand(mine, show, x, y, r, c);
                    win = 0;
                    int i, j;
                    for (i = 1; i <= r; i++) {
                        for (j = 1; j <= c; j++) {
                            if (show[i][j] >= '0' && show[i][j] <= '8')
                                win++;
                        }
                    }
                    Displayboard(show, ROWS, COLS, 1, 1);
                }
            } else {
                printf("该位置已被排查，请重新输入坐标！\n");
            }
        }
    }
    
    if (win >= r * c - count) {
        printf("恭喜你！你不怕危险，敢于排除万难取得胜利！\n");
        int i, j;
        for (i = 1; i <= r; i++) {
            for (j = 1; j <= c; j++) {
                if (mine[i][j] == '1')
                    show[i][j] = '!';
            }
        }
        Displayboard(show, ROWS, COLS, 1, 1);
    }
}
```
