# 一、分支语句
### if语句
+ 单分支：
 仅有if(判断语句){判断为是，执行表达式}
+ 多分支：
    if（情况一判断）
else  if（情况二判断）
else  if（情况三判断）
·
·
·
else       除去上述情况之外的情况
```c
#include <stdio.h>
int main()        //将三个数按照大小排列
{
    int a;
    int b;
    int c;
    scanf("%d%d%d",&a, &b,&c);
    if(a<b)          //判断a<b，如是则运行{}内语句
    {
        int tmp = a;  //确保a永远取最大值
        a = b;
        b = tmp;      
    }
    if(a<c)
    {
        int tmp = a;
        a = c;
        c = tmp;       //用c接收最小值
    }
    if(b<c)
    {
        int tmp = b;
        b = c;
        c = tmp;
    }
    printf("a=%d b=%d c=%d\n", a, b, c);
    return 0;
}
```
### switch语句
+ 格式：
switch（整型或字符型常量与变量）
{
case 值1：表达式；          //当switch（）中值与某个case值相同时执行其后所有语
case 值2：表达式；          //句
·
·
·
default；
}
```c
  #include <stdio.h>
  int main() {
  	int x = 3;
  	int y = 3;
  	switch (x % 2) {  // x%2的结果为1，因此执行case1
  	case 1:
  		switch (y)   // y是3，因此会执行case3，而case3不存在，那只能执行default
  		{
  		case 0:
  			printf("first");
  		case 1:
  			printf("second");
  			break;
  		default: printf("hello"); // 打印hello后，内部switch结束，此时外部case1结束
  		}             // 因为外部case1之后没有添加break语句，所以继续执行case2
  	case 2:             // 打印third
  		printf("third");      // 外部switch结束
  	}
  	return 0;
  }
```
# 二、循环语句
### while语句
+ while(判断语句){循环语句}

```c
#include<stdio.h>
int main()
{
	int a, b, r = 0;
	scanf_s("%d%d", &a, &b);
	r = a % b;      //辗转相除求最大公约数
	while (r!=0) {   //判断r是否为0，如非则循环
		r = a % b;
		a = b;
		b = r;
	}
	printf("%d", a);
	return 0;
}
```

### for语句
+ for(初始化；判断；调整) {循环语句}

```c
#include<stdio.h>
int main()
{
	int i = 0, j = 0;    //99乘法表
	for ( i = 1; i <= 9; ++i)
	{
		for ( j = 0; j <= i;++j)
		{
			printf("%d*%d = %2d   ", i, j, i * j);
		}
		printf("\n");
	}
	return 0;
}
```

# 总结
在C语言中分支（if--else，switch）和循环（while，for，do--while）语句是至关重要的，它们并不只能单独运用，更能相互嵌套实现更多情景的运用。例如以下
```c
#include<stdio.h>
#include<stdlib.h>
#include<time.h>
void game()    //定义一个函数进行猜数部分
{
	int ans, lucky_num, i = 3;
	srand((unsigned int)time(NULL));
	lucky_num = rand() % 30 + 1;
	while (i)
	{
		printf("现在有%d次机会，请输入你的幸运日期(1-30)：", i);
		scanf_s("%d", &ans);
		if (ans == lucky_num)
		{
			printf("猜中了！好幸运，今天也要开开心心呀。\n");
			break;
		}
		if (ans > lucky_num)
		{
			printf("猜大了！\n");
			i--;
		}
		else
		{
			printf("猜小了！\n");
			i--;
		}
	}
}
int main()
{
	while (1)
	{
		int ans;
		printf("----  1.play  ----\n");
		printf("----  0.exit  ----\n");
		printf("请输入你的选择：");
		scanf_s("%d", &ans);
		switch (ans)
		{
		case 1:
			printf("正在加载中···\n");
			game();
			break;
		case 0:
			printf("正在退出···\n");
			goto end;
		default:
			printf("输入错误！请重新输入\n");
			break;
		}
	}
	end:
	return 0;
}
```
